        <script>
            function calculateNFA() {

                var originalPrice =
                    parseFloat(document.getElementById('OriginalPrice').value) || 0;

                var downpayment =
                    parseFloat(document.getElementById('Downpayment').value) || 0;

                var nfa = originalPrice - downpayment;

                document.getElementById('NFA').value = nfa.toFixed(2);
            }

            document.getElementById('OriginalPrice')
                .addEventListener('input', calculateNFA);

            document.getElementById('Downpayment')
                .addEventListener('input', calculateNFA);

            calculateNFA(); // Initial calculation for Edit page
        </script>
